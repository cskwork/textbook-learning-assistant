```typescript
const todayStart = (() => {
  const d = new Date(); d.setHours(0,0,0,0); return d.getTime()
})()
const todayCount = useLiveQuery(
  () => user ? db.quizAttempts.where('studentId').equals(user.email)
    .filter(a => a.attemptedAt >= todayStart).count() : 0,
  [user?.email, todayStart]
) ?? 0
```
