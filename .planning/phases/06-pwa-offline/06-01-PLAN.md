    **Step 1: PWA 아이콘 PNG 파일 생성**

    Node.js 스크립트로 SVG를 PNG로 변환하는 방식 대신, `pnpm` + `sharp` 또는 Canvas API를 사용한다.
    가장 단순한 방법: HTML Canvas를 활용한 Node.js 스크립트 실행.

    아래 스크립트를 `/tmp/gen-icons.mjs`로 작성 후 실행:

    ```javascript
    // /tmp/gen-icons.mjs
    // 의존성 없이 순수 Node.js로 PNG 파일 생성 (PNG binary 직접 작성)
    // 파란색(#2563eb) 배경에 흰색 "수" 텍스트 — 교육 앱 아이콘
    import { writeFileSync } from 'fs'

    // PNG 최소 구조: 파란색 단색 PNG 생성 (sharp 없이)
    // 방법: 미리 생성된 base64 인코딩 파란색 PNG 사용
    // 192x192 파란색 PNG (1×1 픽셀을 192×192로 스케일)
    // 실제로는 단색 PNG를 직접 생성

    function createSolidColorPNG(width, height, r, g, b) {
      // PNG 시그니처
      const sig = Buffer.from([137, 80, 78, 71, 13, 10, 26, 10])

      // IHDR 청크
      const ihdr = Buffer.alloc(13)
      ihdr.writeUInt32BE(width, 0)
      ihdr.writeUInt32BE(height, 4)
      ihdr[8] = 8  // bit depth
      ihdr[9] = 2  // RGB color type
      ihdr[10] = 0 // compression
      ihdr[11] = 0 // filter
      ihdr[12] = 0 // interlace

      // 이미지 데이터 (필터 바이트 0 + RGB 픽셀 반복)
      const rowSize = width * 3
      const rawData = Buffer.alloc(height * (rowSize + 1))
      for (let y = 0; y < height; y++) {
        rawData[y * (rowSize + 1)] = 0 // 필터 바이트 None
        for (let x = 0; x < width; x++) {
          const offset = y * (rowSize + 1) + 1 + x * 3
          rawData[offset] = r
          rawData[offset + 1] = g
          rawData[offset + 2] = b
        }
      }

      const { deflateSync } = await import('zlib').then(m => m)
      // 동기 zlib 사용
      const zlib = require('zlib')
      const compressed = zlib.deflateSync(rawData, { level: 1 })

      function crc32(buf) {
        let crc = 0xFFFFFFFF
        for (const byte of buf) {
          crc ^= byte
          for (let i = 0; i < 8; i++) {
            crc = (crc >>> 1) ^ (crc & 1 ? 0xEDB88320 : 0)
          }
        }
        return (crc ^ 0xFFFFFFFF) >>> 0
      }

      function chunk(type, data) {
        const typeBuffer = Buffer.from(type)
        const lenBuffer = Buffer.alloc(4)
        lenBuffer.writeUInt32BE(data.length, 0)
        const crcData = Buffer.concat([typeBuffer, data])
        const crcBuffer = Buffer.alloc(4)
        crcBuffer.writeUInt32BE(crc32(crcData), 0)
        return Buffer.concat([lenBuffer, typeBuffer, data, crcBuffer])
      }

      const iend = chunk('IEND', Buffer.alloc(0))
      return Buffer.concat([sig, chunk('IHDR', ihdr), chunk('IDAT', compressed), iend])
    }
    ```

    위 방식이 복잡하므로, 대신 **더 단순한 방법** 사용:
    `sharp` 패키지를 임시로 사용하거나, 기존 SVG(apps/web/public/vite.svg)를 변환하는 대신
    **Node.js 내장 모듈만으로** 단색 PNG를 직접 생성하는 올바른 스크립트:

    ```bash
    node -e "
    const zlib = require('zlib');
    const fs = require('fs');

    function makePNG(w, h, r, g, b) {
      const sig = Buffer.from([137,80,78,71,13,10,26,10]);
      function crc32(b){let c=0xFFFFFFFF;for(const x of b){c^=x;for(let i=0;i<8;i++)c=(c>>>1)^(c&1?0xEDB88320:0);}return(c^0xFFFFFFFF)>>>0;}
      function chunk(t,d){const tb=Buffer.from(t),lb=Buffer.alloc(4),cb=Buffer.alloc(4);lb.writeUInt32BE(d.length,0);cb.writeUInt32BE(crc32(Buffer.concat([tb,d])),0);return Buffer.concat([lb,tb,d,cb]);}
      const ihdr=Buffer.alloc(13);ihdr.writeUInt32BE(w,0);ihdr.writeUInt32BE(h,4);ihdr[8]=8;ihdr[9]=2;
      const row=w*3,raw=Buffer.alloc(h*(row+1));
      for(let y=0;y<h;y++){raw[y*(row+1)]=0;for(let x=0;x<w;x++){const o=y*(row+1)+1+x*3;raw[o]=r;raw[o+1]=g;raw[o+2]=b;}}
      const comp=zlib.deflateSync(raw,{level:1});
      return Buffer.concat([sig,chunk('IHDR',ihdr),chunk('IDAT',comp),chunk('IEND',Buffer.alloc(0))]);
    }

    // #2563eb = rgb(37, 99, 235)
    fs.writeFileSync('apps/web/public/pwa-192x192.png', makePNG(192,192,37,99,235));
    fs.writeFileSync('apps/web/public/pwa-512x512.png', makePNG(512,512,37,99,235));
    fs.writeFileSync('apps/web/public/apple-touch-icon.png', makePNG(180,180,37,99,235));
    console.log('아이콘 생성 완료');
    "
    ```

    위 명령을 ` 디렉토리에서 실행.
