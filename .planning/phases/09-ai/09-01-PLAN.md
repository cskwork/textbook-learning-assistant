  2. `const ai = new GoogleGenAI({ apiKey })`
  3. `const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [{ role: 'user', parts: [{ text: userPrompt }] }], config: { systemInstruction: SYSTEM_PROMPT, responseMimeType: 'application/json', responseJsonSchema: MATH_QUESTION_SCHEMA, temperature: 0.7, maxOutputTokens: 2048 } })`

  5. `return JSON.parse(response.text) as GeneratedQuestion`

export async function getGeminiApiKey(userId: string): Promise<string | undefined> {
  const setting = await db.userSettings.where('userId').equals(userId).first()
  return setting?.geminiApiKey

/** Gemini API 키 저장 */
export async function saveGeminiApiKey(userId: string, apiKey: string): Promise<void> {
  const existing = await db.userSettings.where('userId').equals(userId).first()

      isDiagnosisCompleted: false,
      geminiApiKey: apiKey,
    } as UserSetting)
  }
}
