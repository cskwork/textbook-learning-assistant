   ```typescript
   function handleGamificationResult(result: { leveledUp: boolean; newLevel: number; unlockedBadges: string[] }) {
     if (result.leveledUp) {
       setLevelUpInfo({ newLevel: result.newLevel })
     }
     if (result.unlockedBadges.length > 0) {
       // 첫 번째 뱃지만 표시 (여러 개면 순차적으로 — 간단 구현)
       const badgeDef = BADGE_DEFINITIONS.find(b => b.id === result.unlockedBadges[0])
       if (badgeDef) {
         // 레벨업이 있으면 레벨업 후에 뱃지 표시 (setTimeout 3초)
         if (result.leveledUp) {
           setTimeout(() => setUnlockedBadge(badgeDef), 3000)
         } else {
           setUnlockedBadge(badgeDef)
         }
       }
     }
   }
   ```
